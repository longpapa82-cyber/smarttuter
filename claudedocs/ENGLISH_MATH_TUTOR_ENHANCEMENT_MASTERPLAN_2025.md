class AdaptiveDifficultyEngine {
  // 실시간 난이도 조절
  adjustDifficulty(
    studentPerformance: PerformanceMetrics,
    currentLevel: number
  ): number {
    const accuracy = studentPerformance.accuracy;
    const responseTime = studentPerformance.avgResponseTime;

    // 정확도 85% 이상 → 난이도 상승
    if (accuracy > 0.85 && responseTime < targetTime) {
      return Math.min(currentLevel + 1, maxLevel);
    }

    // 정확도 60% 이하 → 난이도 하락
    if (accuracy < 0.6) {
      return Math.max(currentLevel - 1, minLevel);
    }

    return currentLevel;
  }
}

// lib/learning/adaptive-difficulty.ts
export class AdaptiveDifficultyEngine {
  calculateNextDifficulty(history: QuestionHistory[]): DifficultyLevel {
    const recentAccuracy = this.getRecentAccuracy(history, 5);
    const avgResponseTime = this.getAvgResponseTime(history, 5);

    // 스무스한 난이도 조절 (급격한 변화 방지)
    if (recentAccuracy > 0.85 && avgResponseTime < this.targetTime) {
      return this.incrementDifficulty();
    } else if (recentAccuracy < 0.6) {
      return this.decrementDifficulty();
    }

    return this.currentDifficulty;
  }
}

    switch (activity.type) {

    }[activity.difficulty];

## 🎓 교육적 설계 원칙

### 1. Bloom's Taxonomy Integration
```
Level 1 (기억): 개념 정의, 용어 암기
Level 2 (이해): 개념 설명, 예시 제시
Level 3 (적용): 문제 풀이, 실전 적용
Level 4 (분석): 오류 분석, 패턴 인식
Level 5 (평가): 답안 검증, 방법 비교
Level 6 (창조): 문제 생성, 새로운 방법 제안
```

### 2. Constructivist Learning
- 학생이 능동적으로 지식을 구성
- AI는 퍼실리테이터 역할
- 답을 주지 않고 질문으로 유도

### 3. Zone of Proximal Development (ZPD)
- 학생의 현재 수준 파악
- 적절한 도전 과제 제시 (너무 쉽지도, 어렵지도 않게)
- Scaffolding을 통한 점진적 지원
